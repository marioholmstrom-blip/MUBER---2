src/app/components/screens/AssistedBookingScreen.tsx (Flujo de Mudanza Asistida)
Integra el diseño con botones de accesibilidad, selección del nivel de cuidado físico requerido y el botón de dictado de voz gigante.

import { useState } from "react";
import { useNavigate } from "react-router";
import { ArrowLeft, Mic, PhoneCall, Accessibility, Package, UserPlus } from "lucide-react";
import { motion } from "motion/react";

export function AssistedBookingScreen() {
  const navigate = useNavigate();
  const [selectedAssist, setSelectedAssist] = useState<string | null>(null);
  const [isRecording, setIsRecording] = useState(false);

  return (
    <div className="flex h-full w-full flex-col bg-white font-sans text-[#2c3e50]">
      {/* Header */}
      <div className="flex items-center justify-between border-b-8 border-[#2c3e50] bg-[#e67e22] p-6 pt-12">
        <button 
          onClick={() => navigate("/")} 
          className="rounded-xl border-4 border-[#2c3e50] bg-white p-3 shadow-[4px_4px_0px_#2c3e50] transition-transform active:translate-y-1 active:shadow-none"
          aria-label="Volver Atrás"
        >
          <ArrowLeft size={36} strokeWidth={4} />
        </button>
        <h1 className="text-3xl font-black text-[#2c3e50]">
          Asistencia
        </h1>
        {/* Support Button */}
        <button 
          className="rounded-xl border-4 border-white bg-red-600 p-3 shadow-[4px_4px_0px_#2c3e50] transition-transform active:translate-y-1 active:shadow-none text-white"
          aria-label="Ayuda en 1 Clic"
        >
          <PhoneCall size={36} strokeWidth={3} />
        </button>
      </div>

      <div className="flex-1 overflow-y-auto p-6 space-y-8 pb-10">
        
        {/* Voice Command Section */}
        <section className="text-center">
          <h2 className="text-[22px] font-black text-[#2c3e50] mb-6">
            Díganos qué necesita
          </h2>
          <motion.button
            whileTap={{ scale: 0.95 }}
            onClick={() => setIsRecording(!isRecording)}
            className={`group mx-auto flex h-40 w-40 flex-col items-center justify-center rounded-full border-[6px] border-[#2c3e50] shadow-[0_12px_0px_#2c3e50] transition-all active:translate-y-2 active:shadow-none ${
              isRecording ? "bg-red-500 text-white" : "bg-[#e67e22] text-[#2c3e50]"
            }`}
          >
            <Mic size={64} strokeWidth={3} className={isRecording ? "animate-pulse" : ""} />
            <span className="mt-2 text-xl font-bold uppercase tracking-wider">
              {isRecording ? "Escuchando..." : "Hablar"}
            </span>
          </motion.button>
          <p className="mt-8 text-xl font-bold text-slate-600">
            Presione para usar su voz
          </p>
        </section>

        <div className="border-t-4 border-slate-200" />

        {/* Assistance Selection Section */}
        <section>
          <h2 className="text-[22px] font-black text-[#2c3e50] mb-6">
            O seleccione su nivel de ayuda:
          </h2>
          
          <div className="grid grid-cols-1 gap-6">
            {/* Option 1: Wheelchair / Mobility */}
            <button 
              onClick={() => setSelectedAssist("mobility")}
              className={`flex w-full items-center gap-6 rounded-2xl border-[6px] p-6 transition-all shadow-[6px_6px_0px_#2c3e50] active:translate-y-1 active:shadow-none ${
                selectedAssist === "mobility" 
                  ? "border-[#2c3e50] bg-[#e67e22] text-[#2c3e50]" 
                  : "border-slate-300 bg-slate-50 text-slate-700 hover:border-[#2c3e50]"
              }`}
            >
              <div className="rounded-xl border-4 border-current p-3">
                <Accessibility size={48} strokeWidth={2.5} />
              </div>
              <div className="text-left">
                <h3 className="text-2xl font-black mb-1">Movilidad</h3>
                <p className="text-lg font-bold opacity-90">Silla de ruedas o bastón</p>
              </div>
            </button>

            {/* Option 2: Heavy Lifting */}
            <button 
              onClick={() => setSelectedAssist("heavy")}
              className={`flex w-full items-center gap-6 rounded-2xl border-[6px] p-6 transition-all shadow-[6px_6px_0px_#2c3e50] active:translate-y-1 active:shadow-none ${
                selectedAssist === "heavy" 
                  ? "border-[#2c3e50] bg-[#e67e22] text-[#2c3e50]" 
                  : "border-slate-300 bg-slate-50 text-slate-700 hover:border-[#2c3e50]"
              }`}
            >
              <div className="rounded-xl border-4 border-current p-3">
                <Package size={48} strokeWidth={2.5} />
              </div>
              <div className="text-left">
                <h3 className="text-2xl font-black mb-1">Carga Pesada</h3>
                <p className="text-lg font-bold opacity-90">Requiere fuerza física</p>
              </div>
            </button>

            {/* Option 3: Escort / Companion */}
            <button 
              onClick={() => setSelectedAssist("companion")}
              className={`flex w-full items-center gap-6 rounded-2xl border-[6px] p-6 transition-all shadow-[6px_6px_0px_#2c3e50] active:translate-y-1 active:shadow-none ${
                selectedAssist === "companion" 
                  ? "border-[#2c3e50] bg-[#e67e22] text-[#2c3e50]" 
                  : "border-slate-300 bg-slate-50 text-slate-700 hover:border-[#2c3e50]"
              }`}
            >
              <div className="rounded-xl border-4 border-current p-3">
                <UserPlus size={48} strokeWidth={2.5} />
              </div>
              <div className="text-left">
                <h3 className="text-2xl font-black mb-1">Acompañante</h3>
                <p className="text-lg font-bold opacity-90">Ayuda paso a paso</p>
              </div>
            </button>
          </div>
        </section>

        {/* CTA */}
        <button 
          disabled={!selectedAssist}
          className="mt-8 w-full rounded-2xl border-[6px] border-[#2c3e50] bg-[#2c3e50] py-6 text-center text-2xl font-black text-white shadow-[0_12px_0px_#e67e22] transition-transform active:translate-y-2 active:shadow-none disabled:opacity-50 disabled:shadow-none disabled:active:translate-y-0"
        >
          Confirmar Ayuda
        </button>
      </div>
    </div>
  );
}
